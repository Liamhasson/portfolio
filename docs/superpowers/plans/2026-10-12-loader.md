# The Loader Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The story's first arrival in a session opens on the brief: a circle with its construction grows from nothing to the ball's size as the hero's files load. At 100% its points become the hero's own sand, which splits into the vortex cloud while the key light comes up. The copy and nav follow, then scrolling unlocks.

**Architecture:**
- **Pure TS modules:**
  - `loader-mode.ts`: when the loader plays.
  - `load-progress.ts`: honest byte progress.
  - `brief-geometry.ts`: the brief's geometry and timing.
  - `cascade.ts`: the grains' lineage, as GLSL plus a TS mirror.

  Each is unit tested.
- **The brief:** an SVG overlay (`brief.tsx`), outside WebGL.
- **The cascade:** runs in the sand's vertex shader, stateless (like `swirl.ts` and `rising.ts`). An index-based binary lineage over the grains, with their rest spots read from a small integer texture.
- **Orchestration:** `sand-lab.tsx` drives it all from one timeline (`loaderAt`). A boot script in `page.tsx` hides the nav and copy, and locks scrolling before first paint.

**Tech Stack:** Next.js 16 (app router), React 19, three.js 0.186 (GLSL3 ShaderMaterial), Lenis, Vitest + Testing Library, Playwright scripts for captures.

**Spec:** `docs/superpowers/specs/2026-10-12-loader-design.md`.

## Global Constraints

- **No glow anywhere:** no drop-shadow glows, halos, bloom auras or glowing strokes (Liam, 2026-10-12).
- **The brief's style:** warm hairlines `rgb(230, 191, 174)` at 1 CSS px.
  - Anchors are 5 px hollow squares filled with the background `#0b0908`.
  - Guides are dashed at about 30% opacity.
  - Handle length is 0.5523 × the radius; the handle ends are 1.6 px dots.
- **Growth:** the circle grows from 0% to the ball's full size; the size is the progress. It never draws itself on and never shrinks, and takes at least 1.2 s.
- **Anchor order and start times (s after 100%):**

  | Index | Anchor | Starts at |
  |---|---|---|
  | 0 | centre | 0 |
  | 1–4 | curve top, right, bottom, left | 0.32, 0.38, 0.44, 0.5 |
  | 5–8 | box corners TL, TR, BR, BL | 0.2, 0.24, 0.28, 0.3 |

- **Squares:** each anchor's square shrinks away over 0.22 s.
- **The lines dissolve:** they break into dashes from 0.2 s to 1.1 s, and are gone by 1.4 s.
- **The light comes up** from 0.9 s to 2.4 s. Grains are never darker than 35% of their lit level.
- **The copy and nav** fade in from 2.8 s to 3.6 s. Scrolling unlocks at 3.6 s.
- **The brief returns** while the ball forms: it fades in from compaction 0.75, holds, and is gone by 1.0.
- **When it plays:**
  - Once per session (`sessionStorage`).
  - Later arrivals and the deep links `#projects`, `#about` and `#contact` get a 0.6 s fade from black instead.
  - Under reduced motion the brief grows, then a 0.4 s crossfade into the hero with no cascade, and no brief return.
- **Counted in the progress:** the sand (`sand.json`, the three volumes, the tier's parts), the shot's `camera.json`, the haze, the grain tile and the wordmark's font take 90%; compiling and warm-up take the last 10%.
- **Not counted:** the desk. It keeps loading in the background.
- **One centre:** the vortex turns about the ball's centre (`vortexOnBall`, 2026-10-12), so the brief, the vortex and the ball share one centre. The cascade's rest spots include `uChaosShift`, which is that move.
- **The cursor's lean** (`parallax.ts`) stays at rest until the unlock: pass `null` as the cursor to `stepParallax` while `loaderClock !== null`.
- **Budget:** about 5 s in all when cached; 60 fps through the cascade on the high tier.

```ts
// src/app/lab/sand/loader-mode.ts
/**
 * When the loader plays (spec 2026-10-12-loader-design.md §3): once per session, on arrival at the hero; later arrivals
 * and deep links past the hero get a short fade from black; reduced motion: the brief grows, then a crossfade.
 */
export type LoaderMode = "play" | "reduced" | "fade";
export const SEEN_KEY = "liam.loader.seen";
const DEEP = ["#projects", "#about", "#contact"];

export function loaderMode({ seen, hash, reduced }: { seen: boolean; hash: string; reduced: boolean }): LoaderMode {
  if (seen || DEEP.includes(hash)) return "fade";
  return reduced ? "reduced" : "play";
}

export function readSeen(s: Pick<Storage, "getItem"> | null): boolean {
  try { return s?.getItem(SEEN_KEY) === "1"; } catch { return false; }
}

export function markSeen(s: Pick<Storage, "setItem"> | null): void {
  try { s?.setItem(SEEN_KEY, "1"); } catch { /* storage unavailable: it plays again next time */ }
}

/** Runs inline before first paint (page.tsx): the nav and copy stay hidden, and scrolling locked, from the first frame. */
export const LOADER_BOOT = `(function(){try{var s=null;try{s=window.sessionStorage}catch(e){}var seen=false;try{seen=!!s&&s.getItem(${JSON.stringify(SEEN_KEY)})==="1"}catch(e){}var h=location.hash;var deep=${JSON.stringify(DEEP)}.indexOf(h)>=0;var r=window.matchMedia&&window.matchMedia("(prefers-reduced-motion: reduce)").matches;document.documentElement.dataset.loader=(seen||deep)?"fade":(r?"reduced":"play")}catch(e){}})();`;
```

```ts
// src/three/sand/load-progress.ts
/**
 * The loader's progress (spec 2026-10-12-loader-design.md §4): the hero's files by their bytes (90%), then the first
 * frames compiled and warm (10%). Monotonic: a file declared late never sends it backwards.
 */
export class LoadProgress {
  private readonly total = new Map<string, number>();
  private readonly got = new Map<string, number>();
  private compiled = 0;
  private shown = 0;
  constructor(private readonly onChange?: (p: number) => void) {}

  expect(id: string, bytes: number): void { this.total.set(id, Math.max(bytes, 1)); if (!this.got.has(id)) this.got.set(id, 0); this.emit(); }
  add(id: string, bytes: number): void { this.got.set(id, (this.got.get(id) ?? 0) + bytes); this.emit(); }
  done(id: string): void { const t = this.total.get(id) ?? Math.max(this.got.get(id) ?? 1, 1); this.total.set(id, t); this.got.set(id, t); this.emit(); }
  compile(f: number): void { this.compiled = Math.min(Math.max(f, this.compiled), 1); this.emit(); }

  get value(): number { return this.shown; }

  private emit(): void {
    let t = 0, g = 0;
    for (const [id, n] of this.total) { t += n; g += Math.min(this.got.get(id) ?? 0, n); }
    const raw = (t > 0 ? 0.9 * (g / t) : 0) + 0.1 * this.compiled;
    const next = Math.max(this.shown, Math.min(raw, 1));
    if (next !== this.shown) { this.shown = next; this.onChange?.(next); }
  }
}

/** fetch, counting the body as it streams; the returned Response's body is already read (a buffered copy). */
export async function fetchTracked(url: string, id: string, progress: LoadProgress | null): Promise<Response> {
  const res = await fetch(url);
  if (!res.ok) throw new Error(`loader: ${url} ${res.status}`);
  const len = Number(res.headers.get("Content-Length")) || 0;
  if (progress) progress.expect(id, len || 1);
  if (!res.body || !progress) { const buf = await res.arrayBuffer(); progress?.done(id); return new Response(buf, { headers: res.headers }); }
  const reader = res.body.getReader();
  const chunks: Uint8Array[] = [];
  let n = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    chunks.push(value); n += value.byteLength;
    progress.add(id, value.byteLength);
  }
  progress.done(id);
  const out = new Uint8Array(n);
  let o = 0;
  for (const c of chunks) { out.set(c, o); o += c.byteLength; }
  return new Response(out, { headers: res.headers });
}
```

```ts
/** The grains (only the parts needed to draw `grains`, the tier's count) and the density volumes, counted. */
export async function loadSandParts(meta: SandMeta, grains: number, base = SAND_BASE, progress: LoadProgress | null = null): Promise<SandData> {
  const parts = meta.parts.filter((p) => p.start < grains);
  const n = parts.reduce((s, p) => s + p.count, 0);
  const files = ["vol_chaos.bin", "vol_mid.bin", "vol_ball.bin", ...parts.map((p) => p.file)];
  // (declared up front, so the progress knows the whole before the first byte)
  files.forEach((f) => progress?.expect(`${base}/${f}`, 1));
  const [chaos, mid, ball, ...partBufs] = await Promise.all(files.map((f) => bytes(`${base}/${f}`, progress)));
  const words = new Uint32Array(n * 2);
  const attributes = new Uint8Array(n * 4);
  parts.forEach((p, i) => {
    words.set(new Uint32Array(partBufs[i], 0, p.count * 2), p.start * 2);
    attributes.set(new Uint8Array(partBufs[i], p.count * 8, p.count * 4), p.start * 4);
  });
  return {
    meta, count: n, words, attributes,
    volumes: {
      chaos: volumeTexture(new Uint8Array(chaos), meta.volume.res),
      mid: volumeTexture(new Uint8Array(mid), meta.volume.res),
      ball: volumeTexture(new Uint8Array(ball), meta.volume.res),
    },
  };
}

/** Loads the sand: only the parts needed to draw `grains` (the tier's count), and the density volumes. */
export async function loadSand(grains: number, base = SAND_BASE): Promise<SandData> {
  return loadSandParts(await loadSandMeta(base), grains, base);
}
```

```ts
// src/app/lab/sand/brief-geometry.ts
/**
 * The brief (spec 2026-10-12-loader-design.md §2.2, §2.4): the ball's circle with a designer's construction, at the
 * ball's projected centre and radius. It grows from nothing as the hero's files load (dragged out from its centre, as in
 * a design tool), and returns for a beat as the ball locks in on scroll. Screen px.
 */
export const BRIEF = { handle: 0.5523, square: 5, dot: 1.6, rgb: "230, 191, 174", guide: 0.3, minGrow: 1.2 } as const;

export type BriefShapes = {
  cx: number; cy: number; r: number;
  box: { x: number; y: number; s: number };
  /** centre; the curve's top, right, bottom, left; the box's corners TL, TR, BR, BL (the cascade's order) */
  anchors: [number, number][];
  /** each curve anchor's tangent handle: x1, y1, x2, y2 */
  handles: [number, number, number, number][];
};

export function briefShapes(cx: number, cy: number, r: number): BriefShapes {
  const k = BRIEF.handle * r;
  return {
    cx, cy, r,
    box: { x: cx - r, y: cy - r, s: 2 * r },
    anchors: [[cx, cy], [cx, cy - r], [cx + r, cy], [cx, cy + r], [cx - r, cy], [cx - r, cy - r], [cx + r, cy - r], [cx + r, cy + r], [cx - r, cy + r]],
    handles: [[cx - k, cy - r, cx + k, cy - r], [cx + r, cy - k, cx + r, cy + k], [cx - k, cy + r, cx + k, cy + r], [cx - r, cy - k, cx - r, cy + k]],
  };
}

/** The size (0..1 of the ball's): chases the progress (a damped ease), at most the minimum-duration ramp, never back. */
export function briefSize(prev: number, progress: number, elapsed: number, dt: number): number {
  const cap = Math.min(elapsed / BRIEF.minGrow, 1);
  const target = Math.min(progress, cap * cap * (3 - 2 * cap));
  const next = prev + (target - prev) * (1 - Math.exp(-dt * 6));
  return Math.max(prev, Math.min(next, 1), target >= 0.9995 && prev > 0.995 ? 1 : 0);
}

const ease = (x: number) => { const t = Math.min(Math.max(x, 0), 1); return t * t * (3 - 2 * t); };
/** As the ball locks in (compaction 0.75 .. 1): in by 0.85, held to 0.92, gone at 1. */
export function briefReturn(compact: number): number {
  if (compact <= 0.75 || compact >= 1) return 0;
  return compact < 0.85 ? ease((compact - 0.75) / 0.1) : compact <= 0.92 ? 1 : 1 - ease((compact - 0.92) / 0.08);
}
```

```ts
// src/app/lab/sand/loader-timeline.ts
/**
 * The loader from 100% (spec 2026-10-12-loader-design.md §2.3), as a function of the seconds since, accumulated from the
 * frames' dt (a hidden tab pauses it): the anchors become grains, the lines dissolve, the light comes up, the copy and
 * nav arrive, scrolling unlocks.
 */
export const LOADER = {
  squareShrink: 0.22,
  dissolve: [0.2, 1.1] as const,
  briefGone: 1.4,
  light: [0.9, 2.4] as const,
  lightFloor: 0.35,
  copy: [2.8, 3.6] as const,
  unlock: 3.6,
  /** each anchor's moment: centre; the curve's top, right, bottom, left; the box's corners TL, TR, BR, BL */
  starts: [0, 0.32, 0.38, 0.44, 0.5, 0.2, 0.24, 0.28, 0.3],
  reducedFade: 0.4,
  fadeIn: 0.6,
} as const;

const ease = (x: number) => { const t = Math.min(Math.max(x, 0), 1); return t * t * (3 - 2 * t); };
const span = (t: number, [a, b]: readonly [number, number]) => Math.min(Math.max((t - a) / (b - a), 0), 1);

export function loaderAt(t: number) {
  const squares = LOADER.starts.map((s) => 1 - ease((t - s) / LOADER.squareShrink));
  const dash = ease(span(t, LOADER.dissolve));
  const briefAlpha = 1 - ease(span(t, [0.7, LOADER.briefGone]));
  const light = ease(span(t, LOADER.light));
  return {
    squares: squares.map((s) => (s <= 1e-6 ? 0 : s)),
    dash,
    briefAlpha: briefAlpha <= 1e-6 ? 0 : briefAlpha,
    light,
    lightK: LOADER.lightFloor + (1 - LOADER.lightFloor) * light,
    copy: ease(span(t, LOADER.copy)),
    unlocked: t >= LOADER.unlock,
  };
}
```

```ts
// src/three/sand/cascade.ts
import { LOADER } from "@/app/lab/sand/loader-timeline";

/**
 * The cascade (spec 2026-10-12-loader-design.md §5.2): at 100% the brief's 9 anchors become grains, and every grain is
 * born from its parent by splitting, at its parent's place, then travels to its own place in the cloud. A binary lineage
 * over the grains' indices: grain i belongs to anchor i % 9, node j = floor(i / 9) in that anchor's tree (j = 0 the anchor
 * grain), generation floor(log2(j + 1)), parent floor((j - 1) / 2). Stateless: a grain's place is a function of its
 * index, the anchors, the rest spots of itself and its ancestors, and time. The shader computes it (CASCADE_GLSL); this
 * TS mirror is what the tests check.
 */
export type V3 = [number, number, number];

export const CASCADE = {
  anchors: 9,
  gen: 0.15,       // s between a generation and the next
  jitter: 0.1,     // s, each grain's own lateness (< gen: a child is always born after its parent)
  travel: 0.55,    // s from its birth to its place
  impulse: 0.35,   // world units: how far the two halves of a split arc apart
  settle: 0.8,     // s over which the hero's own motion takes over at the end
  maxGen: 18,
} as const;

const hash = (n: number) => { const x = Math.sin(n * 12.9898 + 78.233) * 43758.5453; return x - Math.floor(x); };
const smoother = (x: number) => { const t = Math.min(Math.max(x, 0), 1); return t * t * t * (t * (t * 6 - 15) + 10); };
const genOf = (j: number) => Math.floor(Math.log2(j + 1) + 1e-9);
const birthOf = (i: number) => { const a = i % 9, j = Math.floor(i / 9); return LOADER.starts[a] + genOf(j) * CASCADE.gen + (j > 0 ? CASCADE.jitter * hash(i) : 0); };

export function cascadeEnd(n: number): number {
  return Math.max(...LOADER.starts) + genOf(Math.ceil(n / 9)) * CASCADE.gen + CASCADE.jitter + CASCADE.travel;
}

/** One hop: from where it was born to its own place, arcing to its side (the split), at time t. */
function hop(born: V3, home: V3, side: number, age: number): V3 {
  const u = Math.min(Math.max(age / CASCADE.travel, 0), 1), e = smoother(u);
  const d: V3 = [home[0] - born[0], home[1] - born[1], home[2] - born[2]];
  // the arc's direction: across the path, in the screen's plane (z toward the camera)
  let px = -d[1], py = d[0]; const pl = Math.hypot(px, py) || 1; px /= pl; py /= pl;
  const arc = Math.sin(Math.PI * u) * CASCADE.impulse * side;
  return [born[0] + d[0] * e + px * arc, born[1] + d[1] * e + py * arc, born[2] + d[2] * e];
}

export function cascadeAt(i: number, t: number, anchors: V3[], rest: (i: number) => V3): { born: boolean; pos: V3 } {
  const a = i % 9, j = Math.floor(i / 9), g = genOf(j);
  if (t < birthOf(i)) return { born: false, pos: anchors[a] };
  // walk the lineage from the anchor down: each ancestor's place at its child's birth is where the child is born
  let born: V3 = anchors[a];
  for (let k = 0; k <= g; k++) {
    const jk = ((j + 1) >> (g - k)) - 1, ik = jk * 9 + a;
    const side = jk === 0 ? 0 : jk % 2 === 1 ? 1 : -1;
    if (k === g) return { born: true, pos: hop(born, rest(ik), side, t - birthOf(ik)) };
    const jn = ((j + 1) >> (g - k - 1)) - 1, inext = jn * 9 + a;
    born = hop(born, rest(ik), side, birthOf(inext) - birthOf(ik));
  }
  return { born: true, pos: rest(i) };
}

/** The brief's anchors in the world: on the plane through the ball's centre facing the camera (right/up: the camera's). */
export function anchorsWorld(c: V3, r: number, right: V3, up: V3): V3[] {
  const at = (x: number, y: number): V3 => [c[0] + right[0] * x * r + up[0] * y * r, c[1] + right[1] * x * r + up[1] * y * r, c[2] + right[2] * x * r + up[2] * y * r];
  return [at(0, 0), at(0, 1), at(1, 0), at(0, -1), at(-1, 0), at(-1, 1), at(1, 1), at(1, -1), at(-1, -1)];
}

const f = (v: number) => v.toFixed(4);
export const CASCADE_GLSL = /* glsl */ `
uniform vec3 uAnchors[9];
const float CA_STARTS[9] = float[9](${LOADER.starts.map(f).join(", ")});
```
